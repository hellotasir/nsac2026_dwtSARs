  // Split View & Telemetry
  const [splitView, setSplitView] = useState(false);
  const [leafletLoaded, setLeafletLoaded] = useState(false);

  const mapContainerRef = useRef<HTMLDivElement>(null);
  const mapInstanceRef = useRef<LeafletMap | null>(null);
  const tileLayerRef = useRef<LeafletTileLayer | null>(null);
  const sarOverlayRefT1 = useRef<LeafletImageOverlay | null>(null);
  const sarOverlayRefT2 = useRef<LeafletImageOverlay | null>(null);
  const handleMapClickRef = useRef<
    ((lat: number, lng: number) => Promise<void>) | null
  >(null);

  const handleMapClick = async (lat: number, lng: number) => {
    const latFixed = parseFloat(lat.toFixed(4));
    const lngFixed = parseFloat(lng.toFixed(4));

    let locationName = `Selected Target (${latFixed}°, ${lngFixed}°)`;
    try {
      const res = await fetch(
        `https://nominatim.openstreetmap.org/reverse?format=json&lat=${latFixed}&lon=${lngFixed}`,
      );
      const data = (await res.json()) as NominatimReverseResult;
      if (data && data.display_name) {
        const parts = data.display_name.split(",");
        locationName = `${parts[0].trim()}${parts[1] ? ", " + parts[1].trim() : ""}`;
      }
    } catch {
      // Fallback location name
    }

    const newSpot = createSyntheticHotspot(
      locationName,
      latFixed,
      lngFixed,
      "Map Clicked Target",
    );

    setHotspots((prev) => [
      newSpot,
      ...prev.filter((h) => h.id !== newSpot.id),
    ]);
    setSelectedSpot(newSpot);
  };

  useEffect(() => {
    handleMapClickRef.current = handleMapClick;
  });

  useEffect(() => {
    let animId: number | undefined;
    if (isPlaying) {
      const animate = () => {
        setTimeOffset((prev) => (prev + 0.04 * animSpeed) % (2 * Math.PI * 50));
        animId = requestAnimationFrame(animate);
      };
      animId = requestAnimationFrame(animate);
    }
    return () => {
      if (animId !== undefined) cancelAnimationFrame(animId);
    };
  }, [isPlaying, animSpeed]);

  useEffect(() => {
    if (!document.getElementById("leaflet-css")) {
      const link = document.createElement("link");
      link.id = "leaflet-css";
      link.rel = "stylesheet";
      link.href = "https://unpkg.com/leaflet@1.9.4/dist/leaflet.css";
      document.head.appendChild(link);
    }

    if (window.L) {
      queueMicrotask(() => setLeafletLoaded(true));
    } else {
      const script = document.createElement("script");
      script.src = "https://unpkg.com/leaflet@1.9.4/dist/leaflet.js";
      script.onload = () => setLeafletLoaded(true);
      document.head.appendChild(script);
    }
  }, []);

  useEffect(() => {
    queueMicrotask(() => {
      if (sarMode === "insar_phase") setColormap("phase_wrap");
      else if (sarMode === "backscatter") setColormap("viridis");
      else if (sarMode === "coherence") setColormap("magma");
      else if (sarMode === "rgb_composite") setColormap("jet");
    });
  }, [sarMode]);

  const generateSarDataUrl = useCallback(
    (isPost = false, tOffset = timeOffset) => {
      const canvas = document.createElement("canvas");
      canvas.width = 500;
      canvas.height = 500;
      const ctx = canvas.getContext("2d");
      if (!ctx) return "";
      const w = canvas.width;
      const h = canvas.height;

      const imgData = ctx.createImageData(w, h);
      const data = imgData.data;

      const palette =
        COLOR_PALETTES[colormap]?.colors || COLOR_PALETTES.viridis.colors;
      const paletteLen = palette.length;

      const centerX = w / 2;
      const centerY = h / 2;

      const hexToRgb = (hex: string) => {
        const clean = hex.replace("#", "");
        return {
          r: parseInt(clean.substring(0, 2), 16),
          g: parseInt(clean.substring(2, 4), 16),
          b: parseInt(clean.substring(4, 6), 16),
        };
      };

      const parsedPalette = palette.map(hexToRgb);

      for (let y = 0; y < h; y++) {
        for (let x = 0; x < w; x++) {
          const pixelIdx = (y * w + x) * 4;

          const dx = (x - centerX) / (w / 2);
          const dy = (y - centerY) / (h / 2);
          const dist = Math.sqrt(dx * dx + dy * dy);

          const speckle = speckleFilter
            ? (Math.random() - 0.5) * 0.08
            : (Math.random() - 0.5) * 0.25;

          let value: number;

          if (sarMode === "insar_phase") {
            const deformationMult = isPost
              ? fringeScale * 1.8
              : fringeScale * 0.8;
            const phaseValue = Math.sin(
              dist * deformationMult +
                Math.atan2(dy, dx) * 2 +
                tOffset +
                speckle * 3,
            );
            value = (phaseValue + 1) / 2;
          } else if (sarMode === "backscatter") {
            const wavelengthFactor = radarBand === "L-band" ? 1.0 : 1.4;
            const polFactor =
              polarization === "HV" || polarization === "VH" ? 0.6 : 1.0;
            const baseBg =
              (1 - Math.min(1, dist)) * wavelengthFactor * polFactor;
            const eventDrop =
              isPost && selectedSpot.id === "brahmaputra" ? 0.2 : 1.0;
            value = Math.max(0, Math.min(1, (baseBg + speckle) * eventDrop));
          } else if (sarMode === "coherence") {
            const decay = isPost ? dist * 0.9 : dist * 0.3;
            value = Math.max(0.1, Math.min(0.98, 1 - decay + speckle * 0.5));
          } else {
            const rComp = Math.sin(dx * 4 + tOffset) * 0.5 + 0.5;
            const gComp = Math.cos(dy * 4 + tOffset) * 0.5 + 0.5;
            value = (rComp + gComp) / 2 + speckle;
          }

          const calculatedDb = (value - 1) * 35;
          if (calculatedDb < dbCutoff) {
            data[pixelIdx + 3] = 0;
            continue;
          }

          const colorIdx = Math.min(
            paletteLen - 1,
            Math.floor(value * paletteLen),
          );
          const color = parsedPalette[colorIdx] || parsedPalette[0];

          data[pixelIdx] = color.r;
          data[pixelIdx + 1] = color.g;
          data[pixelIdx + 2] = color.b;
          data[pixelIdx + 3] = Math.floor(opacity * 255);
        }
      }

      ctx.putImageData(imgData, 0, 0);
      return canvas.toDataURL();
    },
    [
      colormap,
      speckleFilter,
      sarMode,
      fringeScale,
      radarBand,
      polarization,
      selectedSpot.id,
      dbCutoff,
      opacity,
      timeOffset,
    ],
  );

  useEffect(() => {
    if (!leafletLoaded || !mapContainerRef.current) return;
    const L = window.L;

    if (!mapInstanceRef.current) {
      const map = L.map(mapContainerRef.current, {
        zoomControl: false,
        attributionControl: false,
      }).setView([selectedSpot.lat, selectedSpot.lng], selectedSpot.zoom);

      L.control.zoom({ position: "bottomright" }).addTo(map);

      tileLayerRef.current = L.tileLayer(selectedBasemap.url, {
        maxZoom: 18,
        attribution: selectedBasemap.attribution,
      }).addTo(map);

      map.on("click", (e: LeafletMouseEvent) => {
        if (handleMapClickRef.current) {
          handleMapClickRef.current(e.latlng.lat, e.latlng.lng);
        }
      });

      map.on("mousemove", (e: LeafletMouseEvent) => {
        const { lat, lng } = e.latlng;
        const dLat = lat - selectedSpot.lat;
        const dLng = lng - selectedSpot.lng;
        const dist = Math.sqrt(dLat * dLat + dLng * dLng) * 10;

        const dbVal = (-12 - dist * 12 + (Math.random() - 0.5) * 2).toFixed(1);
        const phaseValNum = ((dist * fringeScale) % (2 * Math.PI)) - Math.PI;
        const phaseVal = phaseValNum.toFixed(2);
        const cohVal = Math.max(0.12, 0.95 - dist * 0.4).toFixed(2);

        setInspectedPixel({
          db: dbVal,
          phase: `${phaseVal} rad (${(phaseValNum * 12).toFixed(1)} mm)`,
          coherence: cohVal,
          lat: lat.toFixed(4),
          lng: lng.toFixed(4),
        });
      });

      mapInstanceRef.current = map;
    }
  }, [
    fringeScale,
    leafletLoaded,
    selectedBasemap.attribution,
    selectedBasemap.url,
    selectedSpot.lat,
    selectedSpot.lng,
    selectedSpot.zoom,
  ]);

  useEffect(() => {
    if (!mapInstanceRef.current || !leafletLoaded) return;
    const L = window.L;
    const map = mapInstanceRef.current;

    map.flyTo([selectedSpot.lat, selectedSpot.lng], selectedSpot.zoom, {
      duration: 1.5,
    });

    if (tileLayerRef.current) {
      map.removeLayer(tileLayerRef.current);
    }
    tileLayerRef.current = L.tileLayer(selectedBasemap.url, {
      maxZoom: 18,
      attribution: selectedBasemap.attribution,
    }).addTo(map);
  }, [selectedSpot, selectedBasemap, leafletLoaded]);

  useEffect(() => {
    if (!mapInstanceRef.current || !leafletLoaded) return;
    const L = window.L;
    const map = mapInstanceRef.current;

    const spanLat = overlaySize;
    const spanLng = overlaySize;

    const bounds = [
      [selectedSpot.lat - spanLat / 2, selectedSpot.lng - spanLng / 2],
      [selectedSpot.lat + spanLat / 2, selectedSpot.lng + spanLng / 2],
    ];

    if (sarOverlayRefT1.current) map.removeLayer(sarOverlayRefT1.current);
    if (sarOverlayRefT2.current) map.removeLayer(sarOverlayRefT2.current);

    const dataUrlT1 = generateSarDataUrl(false, timeOffset);
    sarOverlayRefT1.current = L.imageOverlay(dataUrlT1, bounds, {
      opacity: opacity,
      interactive: false,
    }).addTo(map);

    if (splitView) {
      const dataUrlT2 = generateSarDataUrl(true, timeOffset);
      sarOverlayRefT2.current = L.imageOverlay(dataUrlT2, bounds, {
        opacity: opacity,
        interactive: false,
      }).addTo(map);
    }
  }, [
    selectedSpot,
    radarBand,
    polarization,
    sarMode,
    colormap,
    opacity,
    overlaySize,
    dbCutoff,
    fringeScale,
    speckleFilter,
    splitView,
    leafletLoaded,
    timeOffset,
    generateSarDataUrl,
  ]);

  const handleLocationSearch = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!searchQuery.trim()) return;

    setIsSearching(true);
    setSearchError("");
    setSearchResults([]);

    try {
      const response = await fetch(
        `https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(
          searchQuery,
        )}&limit=5`,
      );
      const data = (await response.json()) as NominatimSearchResult[];

      if (data && data.length > 0) {
        setSearchResults(data);
      } else {
        setSearchError("No places found matching your query.");
      }
    } catch {
      setSearchError("Geocoding request failed.");
    } finally {
      setIsSearching(false);
    }
  };

  const handleSelectSearchResult = (result: NominatimSearchResult) => {
    const newSpot = createSyntheticHotspot(
      result.display_name.split(",")[0] +
        " (" +
        (result.address?.country || "Custom") +
        ")",
      result.lat,
      result.lon,
      "Geocoded Target",
    );

    setHotspots([newSpot, ...hotspots]);
    setSelectedSpot(newSpot);
    setSearchResults([]);
    setSearchQuery("");
  };

  const handleAddManualLocation = (e: React.FormEvent) => {
    e.preventDefault();
    if (!manualName || !manualLat || !manualLng) return;

    const newSpot = createSyntheticHotspot(
      manualName,
      manualLat,
      manualLng,
      "Manual Lat/Lng Target",
    );

    setHotspots([newSpot, ...hotspots]);
    setSelectedSpot(newSpot);
    setManualName("");
    setManualLat("");
    setManualLng("");
    setShowManualForm(false);
  };

  return (
    <div className="flex flex-col h-screen bg-black text-white font-sans overflow-hidden select-none">
      {/* Header Bar */}
      <header className="h-16 border-b border-neutral-800/80 bg-black/90 backdrop-blur-2xl px-6 flex items-center justify-between shrink-0 z-40">
        {/* Brand Title & Pulse Badge */}
        <div className="flex items-center space-x-3.5">
          <div className="relative">
            <div className="w-9 h-9 rounded-2xl bg-neutral-900 border border-neutral-700/80 flex items-center justify-center shadow-lg">
              <Globe className="w-5 h-5 text-white animate-pulse" />
            </div>
            <span className="absolute -bottom-0.5 -right-0.5 flex h-2.5 w-2.5">
              <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-emerald-400 opacity-75"></span>
              <span className="relative inline-flex rounded-full h-2.5 w-2.5 bg-emerald-500"></span>
            </span>
          </div>
          <div>
            <div className="flex items-center space-x-2">
              <h1 className="font-extrabold text-sm tracking-wider text-white uppercase">
                NISAR Ground Station
              </h1>
              <span className="text-[10px] font-mono px-2 py-0.5 rounded-full bg-neutral-900 text-neutral-300 border border-neutral-700 flex items-center space-x-1">
                <Sparkles className="w-3 h-3 text-cyan-400" />
                <span>{CURRENT_YEAR} Active Science Stream</span>
              </span>
            </div>
            <p className="text-[11px] text-neutral-400">
              NASA-ISRO Synthetic Aperture Radar Operational Surface Visualizer
            </p>
          </div>
        </div>

        {/* Navigation Tabs */}
        <div className="flex items-center bg-neutral-900/90 p-1 rounded-full border border-neutral-800 text-xs font-semibold shadow-inner">
          <button
            onClick={() => setActiveTab("map")}
            className={`flex items-center space-x-2 px-5 py-1.5 rounded-full transition-all duration-200 active:scale-95 ${
              activeTab === "map"
                ? "bg-neutral-800 text-white border border-neutral-600/80 shadow-lg"
                : "bg-neutral-900 border border-neutral-800 text-neutral-300 hover:text-white hover:bg-neutral-800/80"

          >
            <Globe className="w-3.5 h-3.5" />
            <span>Interactive Map</span>
          </button>
          <button
            onClick={() => setActiveTab("analytics")}
            className={`flex items-center space-x-2 px-5 py-1.5 rounded-full transition-all duration-200 active:scale-95 ${
              activeTab === "analytics"
                ? "bg-neutral-800 text-white border border-neutral-600/80 shadow-lg"
                : "bg-neutral-900 border border-neutral-800 text-neutral-300 hover:text-white hover:bg-neutral-800/80"
            }`}
          >
            <BarChart2 className="w-3.5 h-3.5" />
            <span>InSAR Analytics</span>
          </button>
          <button
            onClick={() => setActiveTab("pipeline")}
            className={`flex items-center space-x-2 px-5 py-1.5 rounded-full transition-all duration-200 active:scale-95 ${
              activeTab === "pipeline"
                ? "bg-neutral-800 text-white border border-neutral-600/80 shadow-lg"
                : "bg-neutral-900 border border-neutral-800 text-neutral-300 hover:text-white hover:bg-neutral-800/80"
            }`}
          >
            <Code className="w-3.5 h-3.5" />
            <span>STAC API Pipeline</span>
          </button>

        {/* Orbit Telemetry Pills */}
        <div className="hidden lg:flex items-center space-x-3 text-xs font-mono">
          <div className="bg-neutral-900 border border-neutral-800 px-3 py-1.5 rounded-full text-neutral-300 flex items-center space-x-2">
            <span className="w-2 h-2 rounded-full bg-cyan-400"></span>
            <span>Orbit: 12-Day Repeat ({CURRENT_YEAR} Science Ops)</span>
          </div>
        </div>
      </header>

      {/* Main Content Area */}
      <div className="flex-1 flex overflow-hidden relative">
        {activeTab === "map" && (
          <>
            <aside className="w-80 bg-neutral-950 border-r border-neutral-800 flex flex-col z-30 shrink-0 overflow-y-auto">
              <div className="p-4 space-y-5">
                {/* Basemap Selection */}
                <div className="bg-neutral-900 border border-neutral-800 rounded-2xl p-3.5 shadow-xl space-y-2">
                  <div className="flex items-center justify-between text-xs font-bold text-white">
                    <span className="uppercase tracking-wider text-[11px] text-neutral-400">
                      Basemap Layer
                    </span>
                    <Layers className="w-4 h-4 text-neutral-400" />
                  </div>
                  <div className="bg-black border border-neutral-700/80 p-2.5 rounded-xl text-xs font-medium text-white flex items-center justify-between shadow-inner">
                    <span className="flex items-center space-x-2">
                      <Globe className="w-4 h-4 text-cyan-400" />
                      <span>Esri Satellite Imagery</span>
                    </span>
                    <Check className="w-4 h-4 text-emerald-400" />
                  </div>
                </div>

                {/* NISAR Mission Status */}
                <div className="bg-neutral-900 border border-neutral-800 p-3.5 rounded-2xl space-y-1.5 shadow-xl">
                  <div className="flex items-center space-x-2 text-xs font-bold text-white">
                    <Radio className="w-4 h-4 text-emerald-400 animate-pulse" />
                    <span>NASA-ISRO NISAR Operations ({CURRENT_YEAR})</span>
                  </div>
                  <p className="text-[11px] text-neutral-400 leading-relaxed">
                    Utilizing live radar remote sensing streams from the active{" "}
                    <strong>NASA-ISRO Synthetic Aperture Radar (NISAR)</strong>{" "}
                    mission for sub-centimeter global surface deformation
                    tracking.
                  </p>
                </div>

                {/* Search Target Section */}
                <div className="pt-2 border-t border-neutral-800/80 space-y-3">
                  <div className="flex items-center justify-between">
                    <label className="text-xs font-bold uppercase tracking-wider text-neutral-400 flex items-center space-x-1.5">
                      <Search className="w-3.5 h-3.5 text-white" />
                      <span>Search Location</span>
                    </label>
                    <button
                      onClick={() => setShowManualForm(!showManualForm)}
                      className="bg-neutral-900 border border-neutral-700 text-white hover:bg-neutral-800 text-[10px] font-mono px-2.5 py-1 rounded-full flex items-center space-x-1 transition active:scale-95"
                    >
                      <Plus className="w-3 h-3" />
                      <span>
                        {showManualForm ? "Hide Lat/Lng" : "Manual Lat/Lng"}
                      </span>
                    </button>
                  </div>

                  <form
                    onSubmit={handleLocationSearch}
                    className="flex space-x-2"
                  >
                    <input
                      type="text"
                      placeholder="e.g. Mount Vesuvius, Tokyo..."
                      value={searchQuery}
                      onChange={(e) => setSearchQuery(e.target.value)}
                      className="flex-1 bg-black border border-neutral-800 rounded-xl px-3 py-2 text-xs text-white placeholder-neutral-500 focus:outline-none focus:border-neutral-600 transition"
                    />
                    <button
                      type="submit"
                      disabled={isSearching}
                      className="bg-neutral-900 border border-neutral-700 text-white hover:bg-neutral-800 px-3.5 py-2 rounded-xl text-xs font-semibold flex items-center justify-center transition active:scale-95 shrink-0"
                    >
                      {isSearching ? (
                        <Loader2 className="w-3.5 h-3.5 animate-spin" />
                      ) : (
                        <Search className="w-3.5 h-3.5" />
                      )}
                    </button>
                  </form>

                  {searchError && (
                    <div className="text-[11px] text-rose-300 bg-rose-950/40 border border-rose-800 p-2.5 rounded-xl">
                      {searchError}
                    </div>
                  )}

                    <div className="bg-black border border-neutral-800 rounded-2xl p-2 space-y-1 shadow-2xl">
                      <div className="text-[10px] text-neutral-400 font-mono px-2 py-1 border-b border-neutral-800">
                        Geocoded Locations:
                      </div>
                      {searchResults.map((res, index) => (

                          onClick={() => handleSelectSearchResult(res)}
                          className="w-full text-left p-2.5 rounded-xl bg-neutral-900 border border-neutral-800 hover:border-neutral-700 transition text-xs flex items-center justify-between text-white active:scale-95"
                        >
                          <span className="truncate text-white font-medium">
                            {res.display_name}

                          <span className="font-mono text-[9px] text-neutral-400 ml-2 shrink-0">
                            {parseFloat(res.lat).toFixed(2)}°,{" "}
                            {parseFloat(res.lon).toFixed(2)}°
                          </span>

                    <form

                      <div className="text-[10px] text-white font-bold uppercase tracking-wider">
                        Manual Coordinates Input

                        onChange={(e) => setManualName(e.target.value)}
                        className="w-full bg-black border border-neutral-800 rounded-xl px-2.5 py-1.5 text-xs text-white placeholder-neutral-500 focus:outline-none"
                        required

                          step="any"
                          placeholder="Lat (-90 to 90)"
